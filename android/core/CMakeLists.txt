cmake_minimum_required(VERSION 3.22)
project(lunetile_android_core LANGUAGES C CXX)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
if (NOT RUBY_SOURCE_DIR OR NOT RUBY_BUILD_DIR)
  message(FATAL_ERROR "Set RUBY_SOURCE_DIR and RUBY_BUILD_DIR for the Android Ruby build")
endif()
if (NOT RUBY_ENCODINGS_HEADER_DIR)
  set(RUBY_ENCODINGS_HEADER_DIR "${RUBY_BUILD_DIR}")
endif()
if (NOT EXISTS "${RUBY_ENCODINGS_HEADER_DIR}/encdb.h" OR NOT EXISTS "${RUBY_ENCODINGS_HEADER_DIR}/transdb.h")
  message(FATAL_ERROR "Ruby-generated encdb.h and transdb.h are required in RUBY_ENCODINGS_HEADER_DIR")
endif()
add_custom_command(
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/rgss_data_builtin.h"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/embed_core.py" "${CMAKE_CURRENT_BINARY_DIR}/rgss_data_builtin.h"
  DEPENDS embed_core.py saves.rb scripts.rb project.rb lunetile/types.rb lunetile/data.rb lunetile/data_constructors.rb lunetile/text.rb lunetile/integer_map.rb lunetile/rpg_schema.json
  VERBATIM)
add_library(lunetile_android_core STATIC rgss_data.cpp lunetile/types.cpp "${RUBY_SOURCE_DIR}/ext/stringio/stringio.c" "${CMAKE_CURRENT_BINARY_DIR}/rgss_data_builtin.h")
set_source_files_properties("${RUBY_SOURCE_DIR}/ext/stringio/stringio.c" PROPERTIES COMPILE_DEFINITIONS HAVE_TYPE_RB_IO_MODE_T=1)
target_compile_features(lunetile_android_core PRIVATE cxx_std_17)
target_include_directories(lunetile_android_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${RUBY_SOURCE_DIR}/include" "${RUBY_BUILD_DIR}/.ext/include/aarch64-linux-android-android")
file(GLOB encodings CONFIGURE_DEPENDS "${RUBY_SOURCE_DIR}/enc/*.c")
list(FILTER encodings EXCLUDE REGEX "/(ascii|us_ascii|unicode|utf_8|mktable)\\.c$")
file(GLOB transcoders CONFIGURE_DEPENDS "${RUBY_SOURCE_DIR}/enc/trans/*.c")
list(FILTER transcoders EXCLUDE REGEX "/newline\\.c$")
set(enc_init "#include <ruby.h>\nvoid ruby_init_ext(const char*, void (*)(void));\nvoid Init_encdb(void);\nvoid Init_transdb(void);\n")
set(enc_calls "void Init_enc(void) {\nInit_encdb(); rb_provide(\"enc/encdb.so\");\n")
foreach(source IN LISTS encodings transcoders)
  get_filename_component(name "${source}" NAME_WE)
  if(name STREQUAL "encdb" OR name STREQUAL "transdb")
    continue()
  endif()
  if(source MATCHES "/trans/")
    string(APPEND enc_init "void Init_trans_${name}(void);\n")
    string(APPEND enc_calls "ruby_init_ext(\"enc/trans/${name}.so\", Init_trans_${name});\n")
  else()
    string(APPEND enc_init "void Init_${name}(void);\n")
    string(APPEND enc_calls "ruby_init_ext(\"enc/${name}.so\", Init_${name});\n")
  endif()
endforeach()
string(APPEND enc_calls "Init_transdb(); rb_provide(\"enc/trans/transdb.so\");\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/encinit.c" "${enc_init}${enc_calls}")
add_library(lunetile_android_encodings OBJECT "${CMAKE_CURRENT_BINARY_DIR}/encinit.c" ${encodings} ${transcoders})
target_compile_definitions(lunetile_android_encodings PRIVATE EXTSTATIC=1 ONIG_ENC_REGISTER=rb_enc_register)
target_include_directories(lunetile_android_encodings PRIVATE "${RUBY_SOURCE_DIR}/include" "${RUBY_SOURCE_DIR}" "${RUBY_ENCODINGS_HEADER_DIR}" "${RUBY_BUILD_DIR}" "${RUBY_BUILD_DIR}/.ext/include/aarch64-linux-android-android")
if (RUBY_LIBRARY)
  add_executable(lunetile_core_smoke smoke.cpp)
  target_compile_features(lunetile_core_smoke PRIVATE cxx_std_17)
  target_include_directories(lunetile_core_smoke PRIVATE "${RUBY_SOURCE_DIR}/include" "${RUBY_BUILD_DIR}/.ext/include/aarch64-linux-android-android")
  target_link_options(lunetile_core_smoke PRIVATE -static-libstdc++)
  target_sources(lunetile_core_smoke PRIVATE $<TARGET_OBJECTS:lunetile_android_encodings>)
  target_link_libraries(lunetile_core_smoke PRIVATE lunetile_android_core "${RUBY_LIBRARY}" dl m z)
  add_library(lunetile_save_probe SHARED save_probe.cpp)
  target_include_directories(lunetile_save_probe PRIVATE "${RUBY_SOURCE_DIR}/include" "${RUBY_BUILD_DIR}/.ext/include/aarch64-linux-android-android")
  target_sources(lunetile_save_probe PRIVATE $<TARGET_OBJECTS:lunetile_android_encodings>)
  target_link_libraries(lunetile_save_probe PRIVATE lunetile_android_core "${RUBY_LIBRARY}" dl m z)
endif()
